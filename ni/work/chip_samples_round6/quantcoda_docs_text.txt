  const measurementId = "G-JLG9B7P404";
  const cookieDomain = "quantcoda.com";
  const storageKey = "quantcoda.analyticsConsent";
  const readConsent = () => {
    try { return localStorage.getItem(storageKey); } catch (_) { return null; }
  };
  const initialConsent = readConsent();
  // Never send local previews or explicitly marked QA sessions to production GA.
  // Open ?analytics_test=1 before QA; use ?analytics_test=0 to leave QA mode.
  let internalTraffic = false;
  try {
    const qa = new URLSearchParams(location.search).get("analytics_test");
    if (qa === "1") sessionStorage.setItem("quantcoda.analyticsTest", "1");
    if (qa === "0") sessionStorage.removeItem("quantcoda.analyticsTest");
    internalTraffic = sessionStorage.getItem("quantcoda.analyticsTest") === "1";
  } catch (_) { internalTraffic = true; }
  const eligibleHost = location.hostname === cookieDomain && !internalTraffic;
  const disableKey = `ga-disable-${measurementId}`;
  window[disableKey] = !eligibleHost || initialConsent === "denied";
  window.dataLayer = window.dataLayer || [];
  window.gtag = window.gtag || function(){ dataLayer.push(arguments); };
  gtag("consent", "default", {
    analytics_storage: initialConsent === "denied" ? "denied" : "granted",
    ad_storage: "denied",
    ad_user_data: "denied",
    ad_personalization: "denied",
    wait_for_update: 500
  });

  const writeConsent = (value) => {
    try { localStorage.setItem(storageKey, value); } catch (_) {}
  };
  const clearAnalyticsCookies = () => {
    document.cookie.split(";").forEach(cookie => {
      const name = cookie.split("=")[0].trim();
      if (name === "_ga" || name.startsWith("_ga_")) {
        document.cookie = `${name}=; Max-Age=0; path=/; SameSite=Lax`;
        if (location.hostname === cookieDomain || location.hostname.endsWith(`.${cookieDomain}`)) {
          document.cookie = `${name}=; Max-Age=0; path=/; domain=.${cookieDomain}; SameSite=Lax`;
        }
      }
    });
  };
  const loadAnalytics = () => {
    if (!eligibleHost) return;
    window[disableKey] = false;
    gtag("consent", "update", { analytics_storage: "granted" });
    if (document.querySelector(`script[data-ga-id="${measurementId}"]`)) return;
    gtag("js", new Date());
    const campaign = {};
    const search = new URLSearchParams(location.search);
    for (const [utm, key] of Object.entries({utm_source:"campaign_source", utm_medium:"campaign_medium", utm_campaign:"campaign_name"})) {
      const value = search.get(utm);
      if (value && /^[a-zA-Z0-9_.~-]{1,100}$/.test(value)) campaign[key] = value;
    }
    gtag("config", measurementId, {
      ...campaign,
      allow_google_signals: false,
      allow_ad_personalization_signals: false,
      // Account URLs can contain verification or Checkout tokens. Never send them.
      page_location: location.origin + location.pathname,
      page_referrer: (() => { try { const u = new URL(document.referrer); return u.origin + u.pathname; } catch (_) { return ""; } })()
    });
    const script = document.createElement("script");
    script.async = true;
    script.src = `https://www.googletagmanager.com/gtag/js?id=${encodeURIComponent(measurementId)}`;
    script.dataset.gaId = measurementId;
    document.head.appendChild(script);
  };
  const showConsent = () => {
    const banner = document.getElementById("analytics-consent");
    if (banner) banner.hidden = false;
  };
  const hideConsent = () => {
    const banner = document.getElementById("analytics-consent");
    if (banner) banner.hidden = true;
  };

  window.quantcodaSetAnalyticsConsent = value => {
    writeConsent(value);
    if (value === "granted") {
      loadAnalytics();
    } else {
      window[disableKey] = true;
      gtag("consent", "update", { analytics_storage: "denied" });
      clearAnalyticsCookies();
    }
    hideConsent();
  };

  window.quantcodaTrack = (event, params = {}) => {
    if (!eligibleHost || window[disableKey] || readConsent() === "denied") return false;
    const allowed = ["sign_up", "email_verified", "api_key_created", "api_connection_verified", "purchase"];
    if (!allowed.includes(event)) return false;
    gtag("event", event, {...params, transport_type: "beacon"});
    return true;
  };

  if (initialConsent !== "denied") loadAnalytics();
  document.addEventListener("DOMContentLoaded", () => {
    document.querySelectorAll("[data-analytics-consent]").forEach(button => {
      button.addEventListener("click", () => {
        window.quantcodaSetAnalyticsConsent(button.dataset.analyticsConsent);
      });
    });
    document.querySelectorAll("[data-analytics-settings]").forEach(button => {
      button.addEventListener("click", showConsent);
    });
  });
})();

特色因子（Beta）

      
特色因子是登录后使用的按需计算功能。因子值只随当前请求返回，不保存到业务数据库。目前尚未通过公开 REST、Python SDK 或 MCP 提供；如需计算，请前往 
打开特色因子 →

      

        
日期范围
单次最多 31 个自然日，开始日和结束日都计入。

        
请求规模
单次最多 20 个证券代码，并可选择 1–10 个因子。

        
价格口径
支持原始价格（raw）或后复权价格（hfq）。

      

      
页面预览最多显示前 500 行，下载 CSV 包含本次请求的全部结果。结果字段为

、
symbol
、
factor_id
、
factor_version
、
adjustment
 和 
value
；当前版本标记为 
beta-1
。因子仅用于研究，不构成投资建议。
